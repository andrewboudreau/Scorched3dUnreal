# Artillery Tank + Bigger Textured Terrain Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A drivable placeholder artillery tank (aim turret + barrel + power, fire arced shells) on a ~1 km textured, destructible terrain, with an angle/power HUD and a non-hitching carve.

**Architecture:** All gameplay is tracked C++ in the `Scorched3dUnreal` module, building on the merged fly-shoot sandbox. The existing `ACraterTerrainActor::CarveCrater` is optimized to recompute normals only over the affected region and cook collision off the game thread (async), so the bigger terrain stays smooth. A new `ATankPawn` (engine-primitive hull/turret/barrel) becomes the sandbox default pawn; its aiming is driven by a pure, headless-tested `TankAiming` helper and fires the existing `ARocketProjectile`. A pure-C++ `ASandboxHUD` draws angle/power. The triplanar ground material is generated by a tracked in-editor Python script (the only editor-touching step) and picked up by the terrain via a soft path.

**Tech Stack:** Unreal Engine 5.8 C++; GeometryFramework/GeometryScript (`UDynamicMeshComponent`, `FMeshNormals`); `USpringArmComponent`, `UCameraComponent`, `UStaticMeshComponent`; Enhanced Input (runtime-created); `AHUD`/`UCanvas`; `UProjectileMovementComponent`; UE Automation tests (headless); `unreal.MaterialEditingLibrary` (editor Python).

```cpp
void ACraterTerrainActor::CarveCrater(FVector WorldCenter, float Radius)
{
    UDynamicMesh* Ground = GetDynamicMeshComponent()->GetDynamicMesh();
    if (!Ground || Radius <= 0.0f) return;

    // Heightfield deformation — a direct port of the legacy DeformLandscape model:
    // lower each surface vertex within Radius by the sine falloff (CraterMath::DepthAt),
    // clamped to the bedrock floor. The surface stays single-valued, so there is no lid.
    const FTransform Xf = GetActorTransform();
    const FVector LocalCenter = Xf.InverseTransformPosition(WorldCenter);
    const double LocalBedrockZ =
        Xf.InverseTransformPosition(FVector(WorldCenter.X, WorldCenter.Y, BedrockZ)).Z;
    const double R = Radius;
    const double Scale = DepthScale;

    Ground->EditMesh([&](UE::Geometry::FDynamicMesh3& EditMesh)
    {
        // 1) Lower only the vertices inside the crater; remember which actually moved.
        TSet<int32> MovedVerts;
        for (const int32 Vid : EditMesh.VertexIndicesItr())
        {
            FVector3d V = EditMesh.GetVertex(Vid);
            const double Dist = FMath::Sqrt(
                FMath::Square(V.X - LocalCenter.X) + FMath::Square(V.Y - LocalCenter.Y));
            const double Drop = CraterMath::DepthAt(Dist, R, Scale); // 0 outside Radius
            if (Drop <= 0.0) continue;
            const double NewZ = FMath::Max(V.Z - Drop, LocalBedrockZ);
            if (NewZ < V.Z) // skip verts already at/under bedrock (no actual move)
            {
                V.Z = NewZ;
                EditMesh.SetVertex(Vid, V);
                MovedVerts.Add(Vid);
            }
        }

        // 2) Recompute normals ONLY for the triangles touching moved vertices (dirty region),
        //    instead of the whole mesh — so carve cost tracks the crater, not the terrain size.
        TSet<int32> AffectedTris;
        for (const int32 Vid : MovedVerts)
        {
            for (const int32 Tid : EditMesh.VtxTrianglesItr(Vid))
            {
                AffectedTris.Add(Tid);
            }
        }
        if (AffectedTris.Num() > 0)
        {
            UE::Geometry::FMeshNormals::RecomputeOverlayTriNormals(EditMesh, AffectedTris.Array());
        }
    });

    // 3) Rebuild collision. The component has bUseAsyncCooking=true (set in RegenerateGround),
    //    so this cooks off the game thread and does not hitch; collision is briefly stale until
    //    the cook lands, which is fine for turn-paced artillery.
    GetDynamicMeshComponent()->UpdateCollision(/*bOnlyIfPending*/ false);
}
```

> Note: `EditMesh.VtxTrianglesItr(Vid)` enumerates the triangles around a vertex. If it does not compile in this engine build, substitute the array form: `TArray<int32> VtxTris; EditMesh.GetVtxTriangles(Vid, VtxTris); for (int32 Tid : VtxTris) AffectedTris.Add(Tid);`

```cpp
#include "TankAiming.h"

FVector TankAiming::MuzzleDirection(double YawDeg, double ElevationDeg)
{
    const double Yaw  = FMath::DegreesToRadians(YawDeg);
    const double Elev = FMath::DegreesToRadians(ElevationDeg);
    const double CosE = FMath::Cos(Elev);

    // Horizontal component rotated by yaw, vertical component from elevation.
    return FVector(
        CosE * FMath::Cos(Yaw),
        CosE * FMath::Sin(Yaw),
        FMath::Sin(Elev)).GetSafeNormal();
}
```

```cpp
    MappingContext = NewObject<UInputMappingContext>(this);

    auto MakeAxis1D = [this]() -> UInputAction*
    {
        UInputAction* A = NewObject<UInputAction>(this);
        A->ValueType = EInputActionValueType::Axis1D;
        return A;
    };

    TurretAction    = MakeAxis1D();
    ElevationAction = MakeAxis1D();
    PowerAction     = MakeAxis1D();
    ZoomAction      = MakeAxis1D();

    LookAction = NewObject<UInputAction>(this);
    LookAction->ValueType = EInputActionValueType::Axis2D;

    FireAction = NewObject<UInputAction>(this);
    FireAction->ValueType = EInputActionValueType::Boolean;

    // Turret: D = +1 (right), A = -1 (left).
    MappingContext->MapKey(TurretAction, EKeys::D);
    {
        FEnhancedActionKeyMapping& M = MappingContext->MapKey(TurretAction, EKeys::A);
        M.Modifiers.Add(NewObject<UInputModifierNegate>(this));
    }

    // Elevation: W = +1 (raise), S = -1 (lower).
    MappingContext->MapKey(ElevationAction, EKeys::W);
    {
        FEnhancedActionKeyMapping& M = MappingContext->MapKey(ElevationAction, EKeys::S);
        M.Modifiers.Add(NewObject<UInputModifierNegate>(this));
    }

    // Power: E = +1 (more), Q = -1 (less).
    MappingContext->MapKey(PowerAction, EKeys::E);
    {
        FEnhancedActionKeyMapping& M = MappingContext->MapKey(PowerAction, EKeys::Q);
        M.Modifiers.Add(NewObject<UInputModifierNegate>(this));
    }

    // Look: mouse delta orbits the camera.
    MappingContext->MapKey(LookAction, EKeys::Mouse2D);

    // Zoom: mouse wheel.
    MappingContext->MapKey(ZoomAction, EKeys::MouseWheelAxis);

    // Fire: spacebar.
    MappingContext->MapKey(FireAction, EKeys::SpaceBar);
}

void ATankPawn::OnTurret(const FInputActionValue& Value)
{
    const float Dt = GetWorld() ? GetWorld()->GetDeltaSeconds() : 0.0f;
    TurretYaw += Value.Get<float>() * TurretSpeed * Dt;
    ApplyAimToMesh();
}

void ATankPawn::OnElevation(const FInputActionValue& Value)
{
    const float Dt = GetWorld() ? GetWorld()->GetDeltaSeconds() : 0.0f;
    BarrelElevation = FMath::Clamp(
        BarrelElevation + Value.Get<float>() * ElevationSpeed * Dt, MinElevation, MaxElevation);
    ApplyAimToMesh();
}

void ATankPawn::OnPower(const FInputActionValue& Value)
{
    const float Dt = GetWorld() ? GetWorld()->GetDeltaSeconds() : 0.0f;
    Power = FMath::Clamp(Power + Value.Get<float>() * PowerSpeed * Dt, MinPower, MaxPower);
}

void ATankPawn::OnLook(const FInputActionValue& Value)
{
    const FVector2D Axis = Value.Get<FVector2D>();
    AddControllerYawInput(Axis.X * LookSensitivity);
    AddControllerPitchInput(-Axis.Y * LookSensitivity);
}

void ATankPawn::OnZoom(const FInputActionValue& Value)
{
    if (SpringArm)
    {
        SpringArm->TargetArmLength = FMath::Clamp(
            SpringArm->TargetArmLength - Value.Get<float>() * 100.0f, 400.0f, 4000.0f);
    }
}

void ATankPawn::ApplyAimToMesh()
{
    if (Turret) Turret->SetRelativeRotation(FRotator(0.0f, TurretYaw, 0.0f));
    if (Barrel)
    {
        // Base -90 pitch aligns the cylinder to +X; subtract elevation so it tilts upward.
        Barrel->SetRelativeRotation(FRotator(-90.0f + BarrelElevation, 0.0f, 0.0f));
    }
}

void ATankPawn::OnFire(const FInputActionValue& /*Value*/)
{
    UWorld* World = GetWorld();
    if (!World || !ShellClass) return;

    // Authoritative aim: local-frame direction from tested math, rotated into world space.
    const FVector LocalDir = TankAiming::MuzzleDirection(TurretYaw, BarrelElevation);
    const FVector WorldDir = GetActorTransform().TransformVectorNoScale(LocalDir).GetSafeNormal();
    const FVector SpawnLoc = Muzzle ? Muzzle->GetComponentLocation()
                                    : GetActorLocation() + WorldDir * 300.0f;

    FActorSpawnParameters Params;
    Params.Owner = this;
    Params.Instigator = GetInstigator();
    Params.SpawnCollisionHandlingOverride = ESpawnActorCollisionHandlingMethod::AlwaysSpawn;

    ARocketProjectile* Shell = World->SpawnActor<ARocketProjectile>(
        ShellClass, SpawnLoc, WorldDir.Rotation(), Params);
    if (Shell)
    {
        Shell->Launch(WorldDir, Power);
    }
}
```

Create `tools/make_terrain_material.py`. This builds a slope/height auto-material (grass on flats → rock on slopes → snow on peaks) from world-space vertex normal Z and world height — no textures, no UVs, so it needs no asset inputs. Run it once in the editor (Step 5).

```python
"""
Generates /Game/Sandbox/M_Terrain_Auto — a procedural slope/height auto-material for the
sandbox terrain (no textures/UVs required). Run ONCE inside the Unreal editor:
  Window > Python (or the Output Log cmd line set to Python), then:
      exec(open(r"C:/Users/andre/source/repos/Scorched3dUnreal/tools/make_terrain_material.py").read())
Requires the "Python Editor Script Plugin" enabled.

Blend logic (BaseColor):
  slope  = saturate(VertexNormalWS.Z)                  # 1 on flats, ->0 on cliffs
  height = saturate((WorldPos.Z - LowZ) / (HighZ-LowZ))# 0 low, 1 high
  ground = lerp(Rock, Grass, slope)                    # cliffs rock, flats grass
  color  = lerp(ground, Snow, height * slope)          # high flat areas go snowy
"""
import unreal

PKG_PATH = "/Game/Sandbox"
ASSET_NAME = "M_Terrain_Auto"
LOW_Z = 0.0
HIGH_Z = 5000.0  # matches GameMode HillAmplitude; peaks above this read as snow

mel = unreal.MaterialEditingLibrary
tools = unreal.AssetToolsHelpers.get_asset_tools()

# Create (or replace) the material asset.
if unreal.EditorAssetLibrary.does_asset_exist(f"{PKG_PATH}/{ASSET_NAME}"):
    unreal.EditorAssetLibrary.delete_asset(f"{PKG_PATH}/{ASSET_NAME}")
mat = tools.create_asset(ASSET_NAME, PKG_PATH, unreal.Material, unreal.MaterialFactoryNew())

def expr(cls, x, y):
    return mel.create_material_expression(mat, cls, x, y)

# --- slope = saturate(VertexNormalWS.Z) ---
normal = expr(unreal.MaterialExpressionVertexNormalWS, -1300, -200)
slope_mask = expr(unreal.MaterialExpressionComponentMask, -1050, -200)
slope_mask.set_editor_property("r", False)
slope_mask.set_editor_property("g", False)
slope_mask.set_editor_property("b", True)   # Z component
slope_mask.set_editor_property("a", False)
mel.connect_material_expressions(normal, "", slope_mask, "")
slope = expr(unreal.MaterialExpressionSaturate, -850, -200)
mel.connect_material_expressions(slope_mask, "", slope, "")

# --- height = saturate((WorldPos.Z - LOW_Z) / (HIGH_Z - LOW_Z)) ---
wpos = expr(unreal.MaterialExpressionWorldPosition, -1300, 200)
wz = expr(unreal.MaterialExpressionComponentMask, -1050, 200)
wz.set_editor_property("r", False); wz.set_editor_property("g", False)
wz.set_editor_property("b", True);  wz.set_editor_property("a", False)
mel.connect_material_expressions(wpos, "", wz, "")
sub = expr(unreal.MaterialExpressionSubtract, -850, 200)
sub.set_editor_property("const_b", LOW_Z)
mel.connect_material_expressions(wz, "", sub, "A")
div = expr(unreal.MaterialExpressionDivide, -650, 200)
div.set_editor_property("const_b", max(1.0, HIGH_Z - LOW_Z))
mel.connect_material_expressions(sub, "", div, "A")
height = expr(unreal.MaterialExpressionSaturate, -450, 200)
mel.connect_material_expressions(div, "", height, "")

# --- colors ---
def color(r, g, b, x, y):
    c = expr(unreal.MaterialExpressionConstant3Vector, x, y)
    c.set_editor_property("constant", unreal.LinearColor(r, g, b, 1.0))
    return c

grass = color(0.08, 0.30, 0.06, -650, -500)
rock  = color(0.25, 0.22, 0.18, -650, -350)
snow  = color(0.90, 0.92, 0.95, -650, -150)

# ground = lerp(Rock, Grass, slope)
ground = expr(unreal.MaterialExpressionLinearInterpolate, -300, -350)
mel.connect_material_expressions(rock,  "", ground, "A")
mel.connect_material_expressions(grass, "", ground, "B")
mel.connect_material_expressions(slope, "", ground, "Alpha")

# snowAmt = height * slope (only high AND flat areas go snowy)
snow_amt = expr(unreal.MaterialExpressionMultiply, -300, 100)
mel.connect_material_expressions(height, "", snow_amt, "A")
mel.connect_material_expressions(slope,  "", snow_amt, "B")

# color = lerp(ground, Snow, snowAmt)
final = expr(unreal.MaterialExpressionLinearInterpolate, -50, -200)
mel.connect_material_expressions(ground,   "", final, "A")
mel.connect_material_expressions(snow,     "", final, "B")
mel.connect_material_expressions(snow_amt, "", final, "Alpha")

# Hook BaseColor; a little roughness.
mel.connect_material_property(final, "", unreal.MaterialProperty.MP_BASE_COLOR)
rough = expr(unreal.MaterialExpressionConstant, -50, 100)
rough.set_editor_property("r", 0.9)
mel.connect_material_property(rough, "", unreal.MaterialProperty.MP_ROUGHNESS)

mel.recompile_material(mat)
unreal.EditorAssetLibrary.save_asset(f"{PKG_PATH}/{ASSET_NAME}")
unreal.log(f"Created {PKG_PATH}/{ASSET_NAME}")
```
